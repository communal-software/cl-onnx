cmake_minimum_required(VERSION 3.20)
project(cl_onnx C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT ONNXRUNTIME_ROOT)
  message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the ONNX Runtime distribution")
endif()
add_library(cl_onnx SHARED bridge.c)
target_include_directories(cl_onnx PRIVATE "${ONNXRUNTIME_ROOT}/include")
find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(cl_onnx PRIVATE "${ORT_LIBRARY}")
target_compile_options(cl_onnx PRIVATE -Wall -Wextra -Werror)
set_target_properties(cl_onnx PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
